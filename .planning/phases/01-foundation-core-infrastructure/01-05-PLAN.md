```python
from __future__ import annotations
from ddk.adapters.llm.base import Message, LLMCallConfig, LLMResponse

class FakeLLMAdapter:
    """Returns canned responses keyed by system prompt first word. Records all calls."""
    def __init__(self, responses: dict[str, str | dict] | None = None):
        self.responses: dict[str, str | dict] = responses or {}
        self.call_log: list[dict] = []

    async def complete(self, messages: list[Message], config: LLMCallConfig) -> LLMResponse:
        key = self._extract_key(messages)
        self.call_log.append({"key": key, "messages": messages, "config": config})
        if key not in self.responses:
            raise ValueError(f"FakeLLMAdapter: no fixture for key '{key}'")
        content = self.responses[key]
        if isinstance(content, dict):
            import json
            content = json.dumps(content)
        return LLMResponse(content=content, model=config.model, input_tokens=100, output_tokens=50, cost_usd=0.001)

    async def complete_structured(self, messages: list[Message], config: LLMCallConfig, output_schema: type) -> LLMResponse:
        return await self.complete(messages, config)

    def _extract_key(self, messages: list[Message]) -> str:
        for msg in messages:
            if msg.role == "system":
                return msg.content.split()[0].lower()
        return "default"
```

```python
from __future__ import annotations
from pathlib import Path

class InMemoryFilesystem:
    """Dict-backed filesystem. No disk I/O."""
    def __init__(self) -> None:
        self._files: dict[str, str] = {}
        self._binary_files: dict[str, bytes] = {}
        self._dirs: set[str] = set()

    def read_text(self, path: Path) -> str:
        key = str(path)
        if key not in self._files:
            raise FileNotFoundError(f"No such file: {path}")
        return self._files[key]

    def write_text(self, path: Path, content: str) -> None:
        self._files[str(path)] = content
        self._dirs.add(str(path.parent))

    def list_dir(self, path: Path) -> list[Path]:
        prefix = str(path)
        result = []
        seen = set()
        for key in self._files:
            if key.startswith(prefix + "/"):
                relative = key[len(prefix) + 1:]
                top_level = relative.split("/")[0]
                if top_level not in seen:
                    seen.add(top_level)
                    result.append(Path(key).parent if "/" in relative else Path(key))
        return sorted(result)

    def exists(self, path: Path) -> bool:
        key = str(path)
        return key in self._files or key in self._dirs or key in self._binary_files

    def mkdir(self, path: Path, parents: bool = True) -> None:
        self._dirs.add(str(path))
        if parents:
            for parent in path.parents:
                self._dirs.add(str(parent))

    def read_bytes(self, path: Path) -> bytes:
        key = str(path)
        if key not in self._binary_files:
            raise FileNotFoundError(f"No such file: {path}")
        return self._binary_files[key]

    def write_bytes(self, path: Path, content: bytes) -> None:
        self._binary_files[str(path)] = content
        self._dirs.add(str(path.parent))
```

```python
class TempDirSandbox:
    """Uses tempfile.mkdtemp() instead of git worktree. Real file I/O."""
    def __init__(self) -> None:
        self._sandbox_path: Path | None = None

    async def create(self, source_path: Path) -> Path:
        self._sandbox_path = Path(tempfile.mkdtemp(prefix="ddk-sandbox-"))
        # Copy source files to sandbox
        if source_path.exists():
            shutil.copytree(source_path, self._sandbox_path, dirs_exist_ok=True)
        return self._sandbox_path

    async def cleanup(self) -> None:
        if self._sandbox_path and self._sandbox_path.exists():
            shutil.rmtree(self._sandbox_path)
            self._sandbox_path = None

    async def run_command(self, command: list[str], cwd: Path | None = None, timeout: int = 300) -> CommandResult:
        return CommandResult(returncode=0, stdout="OK", stderr="", duration_ms=10)
```

```python
from ddk.adapters.llm.base import Message, LLMCallConfig, LLMResponse

class FakeFallbackChain:
    """Simulates N provider failures then success."""
    def __init__(self, fail_count: int = 1, success_content: str = "OK"):
        self.fail_count = fail_count
        self.success_content = success_content
        self.attempt = 0
        self.call_log: list[dict] = []

    async def complete(self, messages: list[Message], config: LLMCallConfig) -> LLMResponse:
        self.attempt += 1
        self.call_log.append({"attempt": self.attempt, "model": config.model})
        if self.attempt <= self.fail_count:
            raise ConnectionError(f"Provider failed (attempt {self.attempt}/{self.fail_count})")
        return LLMResponse(content=self.success_content, model=config.model)

    async def complete_structured(self, messages: list[Message], config: LLMCallConfig, output_schema: type) -> LLMResponse:
        return await self.complete(messages, config)
```
