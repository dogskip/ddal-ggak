<objective>
Wave 4: Wire all Phase 4 services into LangGraph — thin adapter nodes, pure router, and builder.py extension.

Purpose: This is the integration layer. validate_node and autofix_node are D-05 thin adapters (state read → service call → state write, no business logic). router.py is a pure function with the oscillation bug fix. builder.py adds the validate→autofix back-edge loop to the existing 7-node graph.

CRITICAL — Oscillation hash placement:
`failure_hash_history` is updated in **autofix_node** (before applying the fix), NOT in validate_node.
LangGraph applies node state updates before calling conditional edge functions. If validate_node
appended the hash, route_after_validate would immediately see `cur_hash in seen` as True and
return "exhausted" on the very first cycle — the autofix loop would never run.

Correct flow:
```
Cycle 1:
  validate_node   → finds failures, does NOT touch failure_hash_history
  route_after_validate: seen=[], h1 not in [] → "fix_available"
  autofix_node    → appends h1 to failure_hash_history, applies fix

Cycle 2 (same failures, fix didn't work):
  validate_node   → finds same failures again, does NOT touch failure_hash_history
  route_after_validate: seen=[h1], h1 in [h1] → "exhausted" ✅ oscillation detected
```

```python
from ddk.graph.router import route_after_validate
from ddk.models.validation import RuntimeValidationReport, FailureRecord
from ddk.models.visual import VisualRegressionReport

    report=None, vis_report=None,
    budget=3, hash_history=None

    return {

    )
    vis = VisualRegressionReport(
        case_id="tc", reference_path="r.png", generated_path="g.png",

    )
    state = make_state(report=report, vis_report=vis)
    assert route_after_validate(state) == "pass"

def test_route_exhausted_when_budget_zero() -> None:
    report = RuntimeValidationReport(
        case_id="tc", build_success=False, type_check_success=True, lint_success=True,
        failures=[FailureRecord(id="f1", taxonomy_id="FT-BUILD-001", severity="critical", message="err")]
    )
    state = make_state(report=report, budget=0)
    assert route_after_validate(state) == "exhausted"

def test_route_fix_available_when_failures_and_budget() -> None:
    report = RuntimeValidationReport(
        case_id="tc", build_success=False, type_check_success=True, lint_success=True,
        failures=[FailureRecord(id="f1", taxonomy_id="FT-BUILD-001", severity="critical", message="err", file_path="src/App.tsx")]
    )
    state = make_state(report=report, budget=2, hash_history=[])
    assert route_after_validate(state) == "fix_available"

def test_route_exhausted_on_oscillation() -> None:
    """CRITICAL: oscillation uses (taxonomy_id, file_path) hash, not UUID."""
    failures = [
        FailureRecord(id="NEW-UUID-1", taxonomy_id="FT-BUILD-001", severity="critical",
                      message="err", file_path="src/App.tsx")
    ]
    # Compute the expected hash for these taxonomy+path combos
    expected_hash = hash(frozenset(
        (f.taxonomy_id, f.file_path or "") for f in failures
    ))
    report = RuntimeValidationReport(
        case_id="tc", build_success=False, type_check_success=True, lint_success=True,
        failures=failures
    )
    # Hash already in history → oscillation detected
    state = make_state(report=report, budget=2, hash_history=[expected_hash])

def test_route_hash_uses_taxonomy_id_not_uuid() -> None:

    f1 = FailureRecord(id=str(uuid.uuid4()), taxonomy_id="FT-TYPE-001", severity="major",

        failures=[f2]  # Different UUID than f1
    )
    state = make_state(report=report, budget=2, hash_history=[h1])  # h1 (same as h2) already seen
    assert route_after_validate(state) == "exhausted"  # Must detect oscillation
```

```python
Hash tracking is autofix_node's responsibility. If validate_node appended the hash,
route_after_validate would find cur_hash in seen on the very first cycle and return
"exhausted" before autofix ever runs.
"""
from __future__ import annotations
from typing import Any


async def validate_node(state: dict[str, Any], **kwargs: Any) -> dict[str, Any]:
    """D-05 thin adapter. No business logic — delegates entirely to RuntimeValidator.

    Does NOT touch failure_hash_history (oscillation state managed by autofix_node).
    """
    container = kwargs.get("context")

    if container is None or not hasattr(container, "runtime_validator") or container.runtime_validator is None:
        return {
            "terminal_state": "manual-review-required",
            "error_message": "RuntimeValidator not configured in ServiceContainer",
            "current_node": "validate",
        }

    report, vis_report = await container.runtime_validator.run(state)

    result: dict[str, Any] = {
        "runtime_validation_report": report,
        "current_node": "validate",
        # failure_hash_history deliberately NOT updated here
    }
    if vis_report is not None:
        result["visual_regression_report"] = vis_report

    return result
```

### B. src/ddk/graph/nodes/autofix.py

```python
"""autofix_node — thin adapter: state → AutoFixPlanner → state. D-05. VALID-04.

IMPORTANT: autofix_node appends the current failure hash to failure_hash_history
BEFORE applying the fix. This enables oscillation detection on the next validate cycle:
if the fix fails and the same failures recur, route_after_validate will find the hash
already in history and return "exhausted".
"""
from __future__ import annotations
from typing import Any


async def autofix_node(state: dict[str, Any], **kwargs: Any) -> dict[str, Any]:
    """D-05 thin adapter. Appends failure hash, delegates to AutoFixPlanner, decrements budget."""
    container = kwargs.get("context")

    # Guard: artifacts must be present to patch
    if state.get("generated_artifacts") is None:
        return {
            "terminal_state": "manual-review-required",
            "error_message": "autofix_node: generated_artifacts is None — cannot apply patches",
            "current_node": "autofix",
        }

    report = state.get("runtime_validation_report")
    failures = report.failures if report else []

    # Record current failure hash BEFORE applying fix.
    # Uses stable (taxonomy_id, file_path) — NOT failure.id (UUID, regenerated each run).
    cur_hash = hash(frozenset((f.taxonomy_id, f.file_path or "") for f in failures))
    updated_history = list(state.get("failure_hash_history", [])) + [cur_hash]

    attempt_n = len(state.get("fix_attempts", [])) + 1

    if container is None or not hasattr(container, "auto_fix_planner") or container.auto_fix_planner is None:
        return {
            "terminal_state": "manual-review-required",
            "error_message": "AutoFixPlanner not configured in ServiceContainer",
            "current_node": "autofix",
        }

    fix_attempt, patched_artifacts = await container.auto_fix_planner.plan_and_apply(
        failures=failures,
        artifacts=state.get("generated_artifacts"),
        case_id=state["case_id"],
        attempt_number=attempt_n,
    )

    return {
        "generated_artifacts": patched_artifacts,
        "fix_attempts": list(state.get("fix_attempts", [])) + [fix_attempt],
        "retry_budget_remaining": state.get("retry_budget_remaining", 3) - 1,
        "failure_hash_history": updated_history,  # hash recorded here, not in validate_node
        "current_node": "autofix",
    }
```

### C. src/ddk/graph/router.py

```python
"""route_after_validate — pure routing function for LangGraph conditional edge. VALID-04."""
from __future__ import annotations
from typing import Any, Literal


def route_after_validate(
    state: dict[str, Any],
) -> Literal["pass", "fix_available", "exhausted"]:
    """Determine next node after validate.

    Returns:
        "pass"          → all checks pass → END
        "fix_available" → failures exist, budget > 0, no oscillation → autofix
        "exhausted"     → budget=0 OR oscillation detected → END (manual-review)

    CRITICAL: Oscillation detection uses hash(frozenset((f.taxonomy_id, f.file_path or "")))
    NOT hash(frozenset(f.id for f in failures)) — UUIDs are regenerated each run!
    """
    report = state.get("runtime_validation_report")
    vis_report = state.get("visual_regression_report")

    # Check: all passing
    def _all_passing() -> bool:
        if report is None:
            return True
        if not (report.build_success and report.type_check_success and report.lint_success):
            return False
        if report.failures:
            return False
        if vis_report is not None and not vis_report.pass_result:
            return False
        return True

    if _all_passing():
        return "pass"

    # Check: budget exhausted
    budget = state.get("retry_budget_remaining", 0)
    if budget <= 0:
        return "exhausted"

    # Check: oscillation (same failure pattern seen before)
    # CORRECT: use (taxonomy_id, file_path) — NOT failure.id (UUID)
    failures = report.failures if report else []
    cur_hash = hash(frozenset((f.taxonomy_id, f.file_path or "") for f in failures))
    seen: list[int] = state.get("failure_hash_history", [])
    if cur_hash in seen:
        return "exhausted"

    return "fix_available"
```
