```python
from __future__ import annotations
import json
from pathlib import Path

SCHEMA_VERSION = "1.0"


class WorkspaceManager:
    """Manages .ddk/ workspace directory structure.

    .ddk/
      schema-version.json          ← {"version": "1.0"}
      cases/
        C-001/
          input.json
          artifacts/
          reports/
          fix-attempts/
          llm-calls/
          screenshots/
    """

    def __init__(self, project_root: Path) -> None:
        self.project_root = project_root
        self.root = project_root / ".ddk"

    def init(self) -> Path:
        """Idempotent workspace initialization. Returns .ddk/ path."""
        self.root.mkdir(exist_ok=True)
        (self.root / "cases").mkdir(exist_ok=True)

        schema_file = self.root / "schema-version.json"
        if not schema_file.exists():
            schema_file.write_text(
                json.dumps({"version": SCHEMA_VERSION}, indent=2),
                encoding="utf-8",
            )

        return self.root

    def next_case_id(self) -> str:
        """Generate next sequential case ID: C-001, C-002, etc."""
        cases_dir = self.root / "cases"
        if not cases_dir.exists():
            return "C-001"

        existing = [
            d.name for d in cases_dir.iterdir()
            if d.is_dir() and d.name.startswith("C-")
        ]
        n = len(existing) + 1
        return f"C-{n:03d}"

    def case_dir(self, case_id: str) -> Path:
        """Return path to a specific case directory."""
        return self.root / "cases" / case_id

    def init_case(self, case_id: str) -> Path:
        """Create case directory with standard subdirectories."""
        case_path = self.case_dir(case_id)
        case_path.mkdir(parents=True, exist_ok=True)

        for subdir in ["artifacts", "reports", "fix-attempts", "llm-calls", "screenshots"]:
            (case_path / subdir).mkdir(exist_ok=True)

        return case_path

    def get_schema_version(self) -> str:
        """Read schema version from workspace."""
        schema_file = self.root / "schema-version.json"
        if not schema_file.exists():
            return "unknown"
        data = json.loads(schema_file.read_text(encoding="utf-8"))
        return data.get("version", "unknown")
```

```python
from __future__ import annotations
import os
import shutil
import subprocess
import sys
from dataclasses import dataclass, field
from typing import Literal


@dataclass
class CheckResult:
    name: str
    status: Literal["pass", "warn", "fail"]
    detail: str = ""
    fix: str | None = None


class PreflightDoctor:
    """Validates environment prerequisites for ddalggak.

    Checks: Python version, uv, API key, Node.js, git version.
    Per CERT-05 / 14-cli-spec.md §1 ddk preflight.
    """

    def run_checks(self) -> list[CheckResult]:
        """Run all preflight checks and return results."""
        return [
            self._check_python_version(),
            self._check_uv_installed(),
            self._check_api_key(),
            self._check_node_installed(),
            self._check_git_version(),
        ]

    def run_all(self) -> tuple[Literal["pass", "warn", "fail"], list[CheckResult]]:
        """Run all checks and return overall status + individual results."""
        results = self.run_checks()
        if any(r.status == "fail" for r in results):
            return "fail", results
        if any(r.status == "warn" for r in results):
            return "warn", results
        return "pass", results

    def _check_python_version(self) -> CheckResult:
        version = sys.version_info
        version_str = f"{version.major}.{version.minor}.{version.micro}"
        if version >= (3, 12):
            return CheckResult(name="python_version", status="pass", detail=version_str)
        return CheckResult(
            name="python_version",
            status="fail",
            detail=version_str,
            fix="Install Python >= 3.12",
        )

    def _check_uv_installed(self) -> CheckResult:
        if shutil.which("uv"):
            try:
                result = subprocess.run(
                    ["uv", "--version"], capture_output=True, text=True, timeout=10,
                )
                version = result.stdout.strip()
                return CheckResult(name="uv_installed", status="pass", detail=version)
            except (subprocess.TimeoutExpired, OSError):
                pass
        return CheckResult(
            name="uv_installed",
            status="fail",
            detail="not found",
            fix="curl -LsSf https://astral.sh/uv/install.sh | sh",
        )

    def _check_api_key(self) -> CheckResult:
        key = os.environ.get("DDK_ANTHROPIC_API_KEY", "")
        if key:
            return CheckResult(name="anthropic_api_key", status="pass", detail="set")
        return CheckResult(
            name="anthropic_api_key",
            status="warn",
            detail="not set",
            fix="export DDK_ANTHROPIC_API_KEY=sk-ant-...",
        )

    def _check_node_installed(self) -> CheckResult:
        if shutil.which("node"):
            try:
                result = subprocess.run(
                    ["node", "--version"], capture_output=True, text=True, timeout=10,
                )
                version = result.stdout.strip()
                return CheckResult(name="node_installed", status="pass", detail=version)
            except (subprocess.TimeoutExpired, OSError):
                pass
        return CheckResult(
            name="node_installed",
            status="warn",
            detail="not found",
            fix="Install Node.js >= 20",
        )

    def _check_git_version(self) -> CheckResult:
        if shutil.which("git"):
            try:
                result = subprocess.run(
                    ["git", "--version"], capture_output=True, text=True, timeout=10,
                )
                version_str = result.stdout.strip()
                # Parse "git version X.Y.Z"
                parts = version_str.split()
                if len(parts) >= 3:
                    version = parts[2]
                    major_minor = tuple(int(x) for x in version.split(".")[:2])
                    if major_minor >= (2, 30):
                        return CheckResult(name="git_version", status="pass", detail=version)
                    return CheckResult(
                        name="git_version",
                        status="fail",
                        detail=version,
                        fix="Upgrade git to >= 2.30",
                    )
            except (subprocess.TimeoutExpired, OSError, ValueError):
                pass
        return CheckResult(
            name="git_version",
            status="fail",
            detail="not found",
            fix="Install git >= 2.30",
        )
```

```python
"""딸깍 CLI — ddk doctor, ddk init."""
from __future__ import annotations

import sys
from pathlib import Path
from typing import Optional

import typer
from rich.console import Console
from rich.table import Table

from ddk import __version__
from ddk.logging.setup import configure_logging

app = typer.Typer(
    name="ddk",
    help="딸깍 — self-correcting UI generation loop",
    no_args_is_help=True,
)
console = Console()


def version_callback(value: bool) -> None:
    if value:
        console.print(f"ddk {__version__}")
        raise typer.Exit()


@app.callback()
def main(
    version: Optional[bool] = typer.Option(None, "--version", callback=version_callback, is_eager=True),
    log_level: str = typer.Option("INFO", "--log-level", help="DEBUG|INFO|WARNING|ERROR"),
) -> None:
    """Global options."""
    configure_logging(level=log_level)


@app.command()
def doctor(
    verbose: bool = typer.Option(False, "--verbose", "-v", help="Show detailed output"),
) -> None:
    """Run preflight environment checks."""
    from ddk.preflight.doctor import PreflightDoctor

    doc = PreflightDoctor()
    overall, results = doc.run_all()

    table = Table(title="ddk preflight checks")
    table.add_column("Check", style="cyan")
    table.add_column("Status")
    table.add_column("Detail")
    table.add_column("Fix", style="dim")

    for r in results:
        status_style = {"pass": "[green]✓[/green]", "warn": "[yellow]⚠[/yellow]", "fail": "[red]✗[/red]"}
        table.add_row(
            r.name,
            status_style.get(r.status, r.status),
            r.detail,
            r.fix or "",
        )

    console.print(table)

    if overall == "fail":
        console.print("\n[red]Preflight failed.[/red] Fix critical issues above.")
        raise typer.Exit(code=1)
    elif overall == "warn":
        console.print("\n[yellow]Preflight passed with warnings.[/yellow]")
        raise typer.Exit(code=2)
    else:
        console.print("\n[green]All checks passed![/green]")


@app.command(name="init")
def init_workspace(
    repo: Path = typer.Option(Path("."), "--repo", help="Target repo root"),
) -> None:
    """Initialize .ddk/ workspace directory."""
    from ddk.workspace.manager import WorkspaceManager

    wm = WorkspaceManager(repo.resolve())
    workspace_path = wm.init()
    console.print(f"[green]Workspace initialized:[/green] {workspace_path}")
```
