```python
from __future__ import annotations
from typing import Any, Callable


async def _node_wrapper(
    node_fn: Callable[..., Any],
    state: dict[str, Any],
    *args: Any,
    **kwargs: Any,
) -> dict[str, Any]:
    """Error boundary for all graph nodes.

    Catches OS/permission/timeout errors and returns error state.
    Used with functools.partial: workflow.add_node("name", partial(_node_wrapper, real_node))
    """
    try:
        return await node_fn(state, *args, **kwargs)
    except (OSError, PermissionError) as e:
        return {
            "terminal_state": "manual-review-required",
            "error_message": f"OS error in node: {e}",
        }
    except TimeoutError as e:
        return {
            "terminal_state": "manual-review-required",
            "error_message": f"Timeout in node: {e}",
        }
    except Exception as e:
        return {
            "terminal_state": "manual-review-required",
            "error_message": f"Unexpected error: {type(e).__name__}: {e}",
        }
```

```python
from __future__ import annotations
import re
from typing import Any

REDACT_PATTERNS: list[tuple[str, re.Pattern[str]]] = [
    ("api_key",     re.compile(r"(sk-[a-zA-Z0-9]{20,})")),
    ("api_key",     re.compile(r"(sk-ant-[a-zA-Z0-9\-]{20,})")),
    ("api_key",     re.compile(r"(AIza[a-zA-Z0-9\-_]{35})")),
    ("bearer",      re.compile(r"(Bearer\s+[a-zA-Z0-9\-_.~+/]+=*)")),
    ("env_secret",  re.compile(r"([A-Z_]{3,}_(?:KEY|TOKEN|SECRET|PASSWORD|PASS)=\S+)")),
    ("private_key", re.compile(r"(-----BEGIN (?:RSA |EC )?PRIVATE KEY-----[\s\S]*?-----END)", re.DOTALL)),
    ("jwt",         re.compile(r"(eyJ[a-zA-Z0-9_-]{10,}\.[a-zA-Z0-9_-]{10,}\.[a-zA-Z0-9_-]{10,})")),
]

REDACT_PLACEHOLDER = "[REDACTED]"


def redact_string(value: str) -> str:
    """Replace all sensitive patterns in a string with [REDACTED]."""
    for _name, pattern in REDACT_PATTERNS:
        value = pattern.sub(REDACT_PLACEHOLDER, value)
    return value


def redact_dict(data: dict[str, Any], max_depth: int = 10) -> dict[str, Any]:
    """Recursively redact string values in a dictionary."""
    if max_depth <= 0:
        return data
    result: dict[str, Any] = {}
    for key, value in data.items():
        if isinstance(value, str):
            result[key] = redact_string(value)
        elif isinstance(value, dict):
            result[key] = redact_dict(value, max_depth - 1)
        elif isinstance(value, list):
            result[key] = [
                redact_string(item) if isinstance(item, str)
                else redact_dict(item, max_depth - 1) if isinstance(item, dict)
                else item
                for item in value
            ]
        else:
            result[key] = value
    return result
```

```python
from __future__ import annotations
import logging
from typing import Any

import structlog

from ddk.security.redact import redact_string


def RedactingProcessor(
    logger: Any, method_name: str, event_dict: dict[str, Any]
) -> dict[str, Any]:
    """Structlog processor that redacts sensitive values in log entries."""
    for key, value in event_dict.items():
        if isinstance(value, str):
            event_dict[key] = redact_string(value)
    return event_dict


def configure_logging(
    level: str = "INFO",
    log_format: str = "text",
    redact: bool = True,
) -> None:
    """Configure structlog with optional JSON output and secret redaction.

    Args:
        level: Log level (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        log_format: "text" for human-readable, "json" for structured JSON
        redact: Whether to enable secret redaction (default True)
    """
    processors: list[Any] = [
        structlog.stdlib.filter_by_level,
        structlog.stdlib.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
    ]

    if redact:
        processors.append(RedactingProcessor)

    if log_format == "json":
        processors.append(structlog.processors.JSONRenderer())
    else:
        processors.append(structlog.dev.ConsoleRenderer())

    structlog.configure(
        processors=processors,
        wrapper_class=structlog.stdlib.BoundLogger,
        logger_factory=structlog.stdlib.LoggerFactory(),
        cache_logger_on_first_use=True,
    )

    # Set stdlib logging level
    logging.basicConfig(level=getattr(logging, level.upper(), logging.INFO), format="%(message)s")
```
