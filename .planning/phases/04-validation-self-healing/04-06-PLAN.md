```python
def start_dev_server(html_content: str, port: int) -> tuple[http.server.HTTPServer, str]:
    tmpdir = Path(tempfile.mkdtemp())
    (tmpdir / "index.html").write_text(html_content)

    handler = http.server.SimpleHTTPRequestHandler
    server = http.server.HTTPServer(("localhost", port), handler)
    server.directory = str(tmpdir)  # type: ignore[attr-defined]

    class DirectoryHandler(http.server.SimpleHTTPRequestHandler):
        def __init__(self, *args, **kwargs):

            pass

    server = http.server.HTTPServer(("localhost", port), DirectoryHandler)

    t.start()
    return server, f"http://localhost:{port}"
```

```python
"""Phase 4 integration tests — real Playwright + real SSIM.

These tests require:
  - Playwright browser binaries: `uv run playwright install chromium`
  - scikit-image + Pillow installed

Skip in CI: pytest -m "not playwright"
"""
from __future__ import annotations

import http.server
import tempfile
import threading
from pathlib import Path

import numpy as np
import pytest
from PIL import Image


pytestmark = pytest.mark.playwright


# ─────────────────────────────────────────────
# Helper: minimal in-process HTTP server
# ─────────────────────────────────────────────

def _start_html_server(html: str, port: int) -> tuple[http.server.HTTPServer, str]:
    """Serve a single HTML string at http://localhost:{port}/."""
    tmpdir = Path(tempfile.mkdtemp(prefix="ddk-playwright-"))
    (tmpdir / "index.html").write_text(html, encoding="utf-8")

    class SilentHandler(http.server.SimpleHTTPRequestHandler):
        def __init__(self, *args, **kwargs):  # type: ignore[no-untyped-def]
            super().__init__(*args, directory=str(tmpdir), **kwargs)

        def log_message(self, *args: object) -> None:  # silence access logs
            pass

    server = http.server.HTTPServer(("localhost", port), SilentHandler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    return server, f"http://localhost:{port}"


# ─────────────────────────────────────────────
# Test 1: Playwright console error capture
# ─────────────────────────────────────────────

@pytest.mark.asyncio
async def test_playwright_captures_console_errors_from_dev_server() -> None:
    """Real Playwright: verify console errors are captured from a live HTML page.

    Serves a minimal HTML page that intentionally throws a JS error,
    then verifies Playwright captures it.
    """
    html = """<!DOCTYPE html>
<html>
<head><title>Test</title></head>
<body>
<script>
  // Intentionally throw a console error for testing
  console.error("test-error-from-ddk-playwright-integration");
</script>
</body>
</html>"""

    server, url = _start_html_server(html, port=18731)
    try:
        from playwright.async_api import ConsoleMessage, async_playwright  # type: ignore[import]

        captured_errors: list[str] = []

        async with async_playwright() as p:
            browser = await p.chromium.launch(headless=True)
            page = await browser.new_page()

            def on_console(msg: ConsoleMessage) -> None:
                if msg.type == "error":
                    captured_errors.append(msg.text)

            page.on("console", on_console)
            await page.goto(url, wait_until="domcontentloaded")
            await page.wait_for_timeout(1000)
            await browser.close()

        assert len(captured_errors) >= 1, f"Expected ≥1 console error, got: {captured_errors}"
        assert any(
            "test-error-from-ddk-playwright-integration" in e for e in captured_errors
        ), f"Expected test error string in: {captured_errors}"

    finally:
        server.shutdown()


# ─────────────────────────────────────────────
# Test 2: SSIM visual regression
# ─────────────────────────────────────────────

def test_ssim_visual_regression_detects_changes(tmp_path: Path) -> None:
    """Real PIL + scikit-image: SSIM detects visual differences between images.

    Creates two 100×100 test images:
    - reference: solid blue
    - modified: half blue, half red (significant visual change)
    Verifies SSIM score < 0.92 threshold for the modified image,
    and SSIM score ≈ 1.0 for identical images.
    """
    from skimage.metrics import structural_similarity  # type: ignore[import]

    # Create reference image (solid blue 100x100)
    ref_arr = np.zeros((100, 100, 3), dtype=np.uint8)
    ref_arr[:, :, 2] = 255  # Blue channel
    ref_path = tmp_path / "reference.png"
    Image.fromarray(ref_arr).save(str(ref_path))

    # Create identical image → SSIM should be ~1.0
    identical_path = tmp_path / "identical.png"
    Image.fromarray(ref_arr.copy()).save(str(identical_path))

    identical_arr = np.array(Image.open(identical_path).convert("RGB"))
    score_identical, _ = structural_similarity(
        ref_arr, identical_arr,
        channel_axis=2,
        data_range=255,
        full=True,
    )
    assert score_identical > 0.99, f"Identical images should have SSIM ≈ 1.0, got {score_identical}"

    # Create significantly different image (half blue, half red)
    diff_arr = np.zeros((100, 100, 3), dtype=np.uint8)
    diff_arr[:50, :, 2] = 255   # top half: blue
    diff_arr[50:, :, 0] = 255   # bottom half: red
    diff_path = tmp_path / "different.png"
    Image.fromarray(diff_arr).save(str(diff_path))

    diff_read = np.array(Image.open(diff_path).convert("RGB"))
    score_different, _ = structural_similarity(
        ref_arr, diff_read,
        channel_axis=2,
        data_range=255,
        full=True,
    )
    threshold = 0.92
    assert score_different < threshold, (
        f"Different images should have SSIM < {threshold}, got {score_different:.4f}"
    )
```
