Implementation outline:

```python
from __future__ import annotations

import time
from pathlib import Path
from typing import Any

from ddk.adapters.build_runner import BuildRunner
from ddk.adapters.playwright_browser import FailedRequestEntry, PlaywrightAdapter
from ddk.adapters.sandbox.base import SandboxAdapter
from ddk.adapters.screenshot_diff import ScreenshotDiffAdapter
from ddk.models.validation import FailureRecord, RuntimeValidationReport
from ddk.models.visual import VisualRegressionReport
from ddk.services.failure_classifier import FailureClassifier


class RuntimeValidator:
    """Sandbox orchestration service — VALID-01, VALID-02, VALID-03."""

    def __init__(
        self,
        sandbox: SandboxAdapter,
        build_runner: BuildRunner,
        playwright: PlaywrightAdapter,
        screenshot_diff: ScreenshotDiffAdapter,
        classifier: FailureClassifier | None = None,
    ) -> None:
        self.sandbox = sandbox
        self.build_runner = build_runner
        self.playwright = playwright
        self.screenshot_diff = screenshot_diff
        self.classifier = classifier or FailureClassifier()

    async def run(
        self,
        state: dict[str, Any],
        dev_server_url: str | None = None,
        reference_screenshot: Path | None = None,
        generated_screenshot: Path | None = None,
    ) -> tuple[RuntimeValidationReport, VisualRegressionReport | None]:
        case_id: str = state["case_id"]
        failures: list[FailureRecord] = []
        runtime_errors: list[str] = []
        start = time.monotonic()

        build_success = True
        type_check_success = True
        lint_success = True

        # VALID-01: Build + typecheck + lint in sandbox
        artifacts = state.get("generated_artifacts")
        if artifacts is not None:
            # Attempt to get artifacts path
            artifacts_path = Path(getattr(artifacts, "output_dir", "."))
            sandbox_dir = await self.sandbox.create(artifacts_path)
            try:
                # Build
                build_result = await self.build_runner.run(
                    ["npm", "run", "build"], cwd=sandbox_dir
                )
                if build_result.returncode != 0:
                    build_success = False
                    if build_result.stderr:
                        rec = self.classifier.classify(build_result.stderr, file_path=None)
                        failures.append(rec)
                        runtime_errors.append(build_result.stderr[:200])

                # Typecheck (only if build passed)
                if build_success:
                    tc_result = await self.build_runner.run(
                        ["npx", "tsc", "--noEmit"], cwd=sandbox_dir
                    )
                    if tc_result.returncode != 0:
                        type_check_success = False
                        if tc_result.stderr:
                            rec = self.classifier.classify(tc_result.stderr)
                            failures.append(rec)

                # Lint
                lint_result = await self.build_runner.run(
                    ["npx", "eslint", "."], cwd=sandbox_dir
                )
                if lint_result.returncode != 0:
                    lint_success = False
                    if lint_result.stderr:
                        rec = self.classifier.classify(lint_result.stderr)
                        failures.append(rec)
            finally:
                await self.sandbox.cleanup()

        # VALID-02: Playwright console + network
        if dev_server_url is not None:
            console_errors = await self.playwright.get_console_errors(dev_server_url)
            for entry in console_errors:
                if entry.level in ("error", "warn"):
                    runtime_errors.append(entry.message)
                    rec = self.classifier.classify(entry.message)
                    failures.append(rec)

            failed_reqs = await self.playwright.get_failed_requests(
                dev_server_url, correlation_id=case_id
            )
            for req in failed_reqs:
                msg = f"Failed request: {req.url} — {req.failure_text}"
                runtime_errors.append(msg)
                rec = self.classifier.classify(req.failure_text or msg)
                failures.append(rec)

        duration_ms = int((time.monotonic() - start) * 1000)
        report = RuntimeValidationReport(
            case_id=case_id,
            build_success=build_success,
            type_check_success=type_check_success,
            lint_success=lint_success,
            runtime_errors=runtime_errors,
            failures=failures,
            duration_ms=duration_ms,
        )

        # VALID-03: SSIM visual regression
        vis_report: VisualRegressionReport | None = None
        if reference_screenshot is not None and generated_screenshot is not None:
            ssim_result = self.screenshot_diff.compute_ssim(
                reference_screenshot, generated_screenshot
            )
            threshold = 0.92
            vis_report = VisualRegressionReport(
                case_id=case_id,
                reference_path=str(reference_screenshot),
                generated_path=str(generated_screenshot),
                ssim_score=ssim_result.score,
                threshold=threshold,
                pass_result=ssim_result.score >= threshold,
                mismatch_regions=[],
                diff_image_path=str(ssim_result.diff_image_path) if ssim_result.diff_image_path else None,
            )

        return report, vis_report
```
