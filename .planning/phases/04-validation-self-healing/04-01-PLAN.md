### A. tests/fakes/fake_build_runner.py

```python
"""FakeBuildRunner — configurable returncode and stderr for unit tests."""

from __future__ import annotations

from pathlib import Path

from ddk.adapters.sandbox.base import CommandResult


class FakeBuildRunner:
    """Returns configurable CommandResult. Records all run() calls."""

    def __init__(
        self,
        returncode: int = 0,
        stdout: str = "Build OK",
        stderr: str = "",
    ) -> None:
        self.returncode = returncode
        self.stdout = stdout
        self.stderr = stderr
        self.run_log: list[dict[str, object]] = []

    async def run(
        self,
        command: list[str],
        cwd: Path,
        timeout: int = 300,
    ) -> CommandResult:
        self.run_log.append({"command": command, "cwd": str(cwd), "timeout": timeout})
        return CommandResult(
            returncode=self.returncode,
            stdout=self.stdout,
            stderr=self.stderr,
            duration_ms=10,
        )
```

```python
"""FakeSSIMAdapter — returns configurable SSIM score for unit tests."""

from __future__ import annotations

from pathlib import Path

from ddk.adapters.screenshot_diff import SSIMResult


class FakeSSIMAdapter:
    """Returns configurable SSIMResult. Records all compute_ssim() calls."""

    def __init__(
        self,
        ssim_score: float = 0.95,
        diff_path: Path | None = None,
    ) -> None:
        self.ssim_score = ssim_score
        self.diff_path = diff_path
        self.compute_log: list[dict[str, object]] = []

    def compute_ssim(self, reference: Path, generated: Path) -> SSIMResult:
        self.compute_log.append({"reference": str(reference), "generated": str(generated)})
        return SSIMResult(
            score=self.ssim_score,
            mismatch_regions=[],
            diff_image_path=self.diff_path,
        )

    def generate_diff_image(self, reference: Path, generated: Path, output: Path) -> Path:
        return output
```
