```python
"""FailureClassifier — maps raw error strings to FAILURE_TAXONOMY IDs. No I/O. VALID-05."""

from __future__ import annotations

import re
import uuid

from ddk.models.validation import FAILURE_TAXONOMY, FailureRecord

# (pattern, taxonomy_id, severity)
# Ordered: more specific patterns first, broader patterns last.
_RULES: list[tuple[str, str, str]] = [
    (r"Cannot find module|Module not found|Failed to resolve", "FT-BUILD-001", "critical"),
    (r"SyntaxError|Unexpected token|Unexpected end of|Unexpected identifier", "FT-BUILD-002", "critical"),
    (r"Cannot find name|is not assignable to type|Type .* is not", "FT-TYPE-001", "major"),
    (r"implicitly has an .any. type|lacks return.type annotation|Parameter .* implicitly", "FT-TYPE-002", "minor"),
    (r"is not a module|has no exported member|does not provide an export", "FT-BUILD-003", "major"),
    (r"Unused variable|Unused import|is defined but never used|no-unused", "FT-LINT-001", "minor"),
    (r"Expected.*semicolon|Parsing error|Expected .*token|Missing semicolon", "FT-LINT-002", "minor"),
    (r"naming.convention|camelCase|PascalCase|SCREAMING_SNAKE", "FT-LINT-003", "minor"),
    (r"Uncaught TypeError|Cannot read propert|Cannot read properties of null|null is not", "FT-RUNTIME-002", "critical"),
    (r"Unhandled.*rejection|UnhandledPromise|UnhandledRejection|unhandledRejection", "FT-RUNTIME-001", "critical"),
    (r"404|503|Failed to fetch|net::ERR_|ECONNREFUSED|request.*failed", "FT-RUNTIME-003", "major"),
    (r"Layout shift|Cumulative Layout|CLS score", "FT-VISUAL-001", "major"),
    (r"color.*mismatch|mismatch.*color|expected.*color", "FT-VISUAL-002", "minor"),
    (r"element.*missing|missing.*element|not found in DOM", "FT-VISUAL-003", "minor"),
    (r"data.binding|binding.*fail|prop.*undefined|undefined.*prop", "FT-DATA-001", "major"),
    (r"missing.*prop|required.*prop|prop.*required", "FT-DATA-002", "minor"),
    (r"race condition|concurrent|TOCTOU", "FT-ASYNC-001", "major"),
    (r"await.*not.*async|async.*missing.*await|Promise.*not.*await|unresolved.*promise", "FT-ASYNC-002", "major"),
]


class FailureClassifier:
    """Maps raw error strings to FAILURE_TAXONOMY IDs. Pure logic — no I/O."""

    def classify(self, raw_message: str, file_path: str | None = None) -> FailureRecord:
        """Classify a raw error message and return a FailureRecord with stable taxonomy_id.

        Always returns a FailureRecord — never raises, never returns None.
        Unrecognized messages get taxonomy_id='FT-UNKNOWN'.
        """
        for pattern, taxonomy_id, severity in _RULES:
            if re.search(pattern, raw_message, re.IGNORECASE):
                return FailureRecord(
                    id=str(uuid.uuid4()),
                    taxonomy_id=taxonomy_id,
                    severity=severity,  # type: ignore[arg-type]
                    message=raw_message[:500],
                    file_path=file_path,
                    raw_output=raw_message,
                )
        return FailureRecord(
            id=str(uuid.uuid4()),
            taxonomy_id="FT-UNKNOWN",
            severity="major",
            message=raw_message[:500],
            file_path=file_path,
            raw_output=raw_message,
        )
```

### RollbackController Implementation

```python
"""RollbackController — git checkpoint management for auto-fix attempts. VALID-07."""

from __future__ import annotations

from pathlib import Path

from ddk.adapters.git import GitAdapter


class RollbackController:
    """Manages git checkpoint refs for each auto-fix attempt.

    Ref format: refs/ddk/fix/{case_id}/attempt-{n}
    Delegates all git operations to the injected GitAdapter (no direct subprocess).
    """

    def __init__(self, git: GitAdapter, repo_path: Path) -> None:
        self.git = git
        self.repo_path = repo_path

    def _ref_name(self, case_id: str, attempt_number: int) -> str:
        return f"refs/ddk/fix/{case_id}/attempt-{attempt_number}"

    async def create_checkpoint(self, case_id: str, attempt_number: int) -> str:
        """Create a git ref checkpoint. Returns commit hash."""
        ref = self._ref_name(case_id, attempt_number)
        return await self.git.create_checkpoint(self.repo_path, ref)

    async def rollback_to_latest(self, case_id: str, attempt_number: int) -> None:
        """Roll back to the checkpoint for this attempt number."""
        ref = self._ref_name(case_id, attempt_number)
        await self.git.rollback_to(self.repo_path, ref)
```
