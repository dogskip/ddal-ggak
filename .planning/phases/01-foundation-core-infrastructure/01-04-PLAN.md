    content: str

class LLMCallConfig:
    model: str = "claude-3-5-sonnet-20241022"
    temperature: float = 0.0
    max_tokens: int = 4096
    seed: int | None = None
    response_format: type | None = None  # Pydantic model for structured output

@dataclass
class LLMResponse:
    content: str
    model: str
    input_tokens: int = 0
    output_tokens: int = 0
    cost_usd: float = 0.0
    raw_response: dict[str, Any] | None = None


@runtime_checkable
class LLMAdapter(Protocol):
    async def complete(
        self, messages: list[Message], config: LLMCallConfig
    ) -> LLMResponse: ...

    async def complete_structured(
        self,
        messages: list[Message],
        config: LLMCallConfig,
        output_schema: type,

class FilesystemAdapter:
    """Real filesystem implementation."""
    def read_text(self, path: Path) -> str:
        return path.read_text(encoding="utf-8")

    def write_text(self, path: Path, content: str) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(content, encoding="utf-8")

    def list_dir(self, path: Path) -> list[Path]:

```python
from __future__ import annotations
from pathlib import Path


ENV_WHITELIST: frozenset[str] = frozenset({
    "PATH", "HOME", "USER", "LANG", "TERM",
    "NODE_ENV", "npm_config_cache",
    "DDK_SANDBOX_MODE", "DDK_LOG_LEVEL",
})

FORBIDDEN_READ_PATHS: list[str] = [
    "~/.ssh/",
    "~/.aws/",
    "~/.config/gcloud/",
    "~/.gnupg/",
    "/etc/shadow",
    "/etc/passwd",
]

ALLOWED_COMMANDS: list[str] = [
    "npm", "npx", "pnpm", "yarn", "bun",
    "node", "tsc", "vite",
    "git",
]


def is_command_allowed(command: str) -> bool:
    """Check if a command is in the allowlist."""
    base_command = command.split("/")[-1].split()[0] if command else ""
    return base_command in ALLOWED_COMMANDS


def is_path_forbidden(path: str | Path) -> bool:
    """Check if a path matches any forbidden read path."""
    path_str = str(Path(path).expanduser())
    for forbidden in FORBIDDEN_READ_PATHS:
        forbidden_expanded = str(Path(forbidden).expanduser())
        if path_str.startswith(forbidden_expanded):
            return True
    return False


def is_env_allowed(var_name: str) -> bool:
    """Check if an environment variable is in the whitelist."""
    return var_name in ENV_WHITELIST
```
