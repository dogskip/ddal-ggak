```python
"""GenerationPlanner — creates ordered GenerationPlan (INGEST-06)."""

from __future__ import annotations

import json
from typing import Any

from ddk.adapters.llm.base import LLMAdapter, LLMCallConfig, LLMResponse, Message
from ddk.models.generation import GenerationNode, GenerationPlan
from ddk.models.intent import IntentGraph
from ddk.models.repo import RepoProfile

_SYSTEM_PROMPT = """Senior React architect here. Given a UI intent graph and repository profile, create a precise GenerationPlan that describes exactly which files to create or modify.

## Planning Rules

1. **Target files must be specific**: Use exact paths relative to repo root.
2. **Respect repo conventions**: Follow the NamingRules and FileLayoutRules from RepoProfile exactly.
3. **Topological ordering**: List nodes in dependency order (types first, then components, then pages, then tests).
4. **Minimal changes**: Only include files that are necessary. Do not generate test files unless explicitly requested.
5. **Reuse existing**: If an existing component matches, reference it instead of generating a duplicate.

## Node Types
- `type`: TypeScript interface/type definition file
- `component`: React component (.tsx)
- `hook`: React custom hook
- `style`: CSS module or style file
- `route`: Page-level component (Next.js page, React Router route)
- `test`: Test file (.test.tsx)
- `file`: Generic file

Output a complete GenerationPlan JSON with execution_order sorted topologically."""


class GenerationPlanner:
    """Creates topologically ordered GenerationPlan for code generation."""

    def __init__(self, llm: LLMAdapter) -> None:
        self._llm = llm
        self._config = LLMCallConfig(
            model="claude-3-5-sonnet-20241022",
            temperature=0.0,
            max_tokens=3072,
        )

    async def plan(
        self,
        case_id: str,
        intent_graph: IntentGraph,
        repo_profile: RepoProfile,
        user_prompt: str,
    ) -> GenerationPlan:
        """Generate a code generation plan from intent + profile.

        Args:
            case_id: Case identifier.
            intent_graph: Extracted UI intent.
            repo_profile: Repository conventions.
            user_prompt: Original user request.

        Returns:
            GenerationPlan with topologically ordered nodes.
        """
        user_content = (
            f"Create a GenerationPlan for this UI.\n\n"
            f"## User Intent\n{user_prompt}\n\n"
            f"## Intent Graph (summary)\n"
            f"- Layout regions: {len(intent_graph.layout_regions)}\n"
            f"- Components: {len(intent_graph.components)}\n"
            f"- Interactions: {len(intent_graph.interactions)}\n"
            f"- Confidence: {intent_graph.confidence}\n"
            f"{intent_graph.model_dump_json(indent=2)[:3000]}\n\n"
            f"## Repository Profile\n"
            f"{repo_profile.model_dump_json(indent=2)}\n\n"
            f"Output the complete GenerationPlan JSON."
        )

        messages = [
            Message(role="system", content=_SYSTEM_PROMPT),
            Message(role="user", content=user_content),
        ]

        response: LLMResponse = await self._llm.complete_structured(
            messages, self._config, output_schema=GenerationPlan
        )

        data: dict[str, Any] = json.loads(response.content)
        data["case_id"] = case_id
        nodes_data = data.get("nodes", [])
        data.setdefault("execution_order", [n.get("id", "") for n in nodes_data])
        data.setdefault("target_files", list({n.get("target_path", "") for n in nodes_data if n.get("target_path")}))
        data.setdefault("estimated_node_count", len(nodes_data))

        return GenerationPlan.model_validate(data)
```

from __future__ import annotations

from typing import Any

from ddk.graph.context import ServiceContainer
from ddk.services.generation_planner import GenerationPlanner


async def plan_node(state: dict[str, Any], **kwargs: Any) -> dict[str, Any]:
    """Create generation plan from intent graph and repo profile."""
    container: ServiceContainer | None = kwargs.get("context")
    llm = container.llm if container else None
    if llm is None:
        return {"terminal_state": "manual-review-required", "error_message": "No LLM adapter"}
    planner = GenerationPlanner(llm=llm)
    plan = await planner.plan(
        case_id=state["case_id"],
        intent_graph=state["intent_graph"],
        repo_profile=state["repo_profile"],
        user_prompt=state["case_input"].prompt,
    )
    return {
        "generation_plan": plan,
        "current_node": "plan",
    }

```python
"""ContextPackBuilder — deterministic context packaging (CODEGEN-03).

Algorithm from docs/design/24-context-pack-determinism.md:
- sort_key = (priority_rank, relation_rank, line_bucket, normalized_path)
- MAX_FILES=6, MAX_LINES=300
- P1~P5 priority hierarchy
- insufficient_context=True when no P1 files found
"""

from __future__ import annotations

import hashlib
import json
import math
from dataclasses import dataclass
from pathlib import Path
from typing import Literal

from ddk.models.context_pack import ContextChunk, ContextPack

MAX_FILES = 6
MAX_LINES = 300

_PRIORITY_RANK = {"P1": 1, "P2": 2, "P3": 3, "P4": 4, "P5": 5}

_BINARY_EXTENSIONS = frozenset({
    ".png", ".jpg", ".jpeg", ".gif", ".webp", ".ico",
    ".woff", ".woff2", ".ttf", ".eot", ".otf",
    ".mp3", ".mp4", ".wav", ".avi",
    ".zip", ".tar", ".gz",
    ".pdf", ".doc", ".docx",
})

# P1 entry-point patterns
_P1_PATTERNS = frozenset({
    "src/App.tsx", "src/App.jsx", "src/main.tsx", "src/main.jsx",
    "src/index.tsx", "src/index.jsx", "package.json",
})

# P2 config patterns
_P2_PATTERNS = frozenset({
    "tsconfig.json", "vite.config.ts", "vite.config.js",
})


@dataclass
class _Candidate:
    path: str
    priority: Literal["P1", "P2", "P3", "P4", "P5"]
    relation_rank: int  # exact=1, same-dir=2, shared=3
    content: str

    @property
    def line_count(self) -> int:
        return self.content.count("\n") + 1

    @property
    def sort_key(self) -> tuple[int, int, int, str]:
        return (
            _PRIORITY_RANK[self.priority],
            self.relation_rank,
            math.ceil(self.line_count / 50),
            self.path.replace("\\", "/").lower(),
        )


class ContextPackBuilder:
    """Builds deterministic ContextPack with P1-P5 priority, MAX_FILES=6, MAX_LINES=300."""

    def build(
        self,
        target_files: list[str],
        repo_path: Path,
        node_name: str = "generate",
    ) -> ContextPack:
        """Build a context pack for the given target files.

        Args:
            target_files: Files that will be generated/modified.
            repo_path: Repository root path.
            node_name: Pipeline node name (for manifest).

        Returns:
            Deterministic ContextPack.
        """
        candidates = self._gather_candidates(repo_path, target_files)
        candidates.sort(key=lambda c: c.sort_key)

        # Select up to MAX_FILES
        selected = candidates[:MAX_FILES]

        # Drop from end (lowest priority) until within MAX_LINES
        while self._total_lines(selected) > MAX_LINES and selected:
            selected.pop()

        # Check for P1 presence
        has_p1 = any(c.priority == "P1" for c in selected)
        insufficient = not has_p1

        # Build chunks
        chunks = [
            ContextChunk(
                path=c.path,
                priority=c.priority,
                start_line=1,
                end_line=c.line_count,
                content=c.content,
                content_sha256=hashlib.sha256(c.content.encode()).hexdigest(),
            )
            for c in selected
        ]

        # Compute deterministic hash
        pack_hash = self._compute_hash(node_name, target_files, chunks)

        missing_for: list[str] = []
        if insufficient:
            missing_for = target_files

        return ContextPack(
            node_name=node_name,
            target_files=target_files,
            chunks=chunks,
            total_lines=sum(c.end_line - c.start_line + 1 for c in chunks),
            insufficient_context=insufficient,
            missing_context_for=missing_for,
            context_pack_hash=pack_hash,
        )

    def _gather_candidates(
        self, repo_path: Path, target_files: list[str]
    ) -> list[_Candidate]:
        candidates: list[_Candidate] = []
        target_dirs = {str(Path(tf).parent) for tf in target_files}

        for file_path in sorted(repo_path.rglob("*")):
            if not file_path.is_file():
                continue

            rel = str(file_path.relative_to(repo_path)).replace("\\", "/")

            # Exclude node_modules and binary files
            if "node_modules" in rel or "__pycache__" in rel:
                continue
            if file_path.suffix.lower() in _BINARY_EXTENSIONS:
                continue

            try:
                content = file_path.read_text(encoding="utf-8")
            except (UnicodeDecodeError, OSError):
                continue

            priority = self._classify_priority(rel, target_dirs)
            relation_rank = self._classify_relation(rel, target_files, target_dirs)

            candidates.append(_Candidate(
                path=rel,
                priority=priority,
                relation_rank=relation_rank,
                content=content,
            ))

        return candidates

    def _classify_priority(
        self, rel_path: str, target_dirs: set[str]
    ) -> Literal["P1", "P2", "P3", "P4", "P5"]:
        if rel_path in _P1_PATTERNS:
            return "P1"
        if rel_path in _P2_PATTERNS or any(rel_path.startswith(p) for p in ("tsconfig.",)):
            return "P2"
        # P3: source files in src/
        if rel_path.endswith((".tsx", ".jsx", ".ts", ".js")) and rel_path.startswith("src/"):
            parent = str(Path(rel_path).parent)
            if parent in target_dirs:
                return "P2"  # same dir as target → P2 sibling
            return "P3"
        # P4: style files
        if rel_path.endswith((".css", ".scss", ".sass", ".module.css")):
            return "P4"
        # P5: everything else (docs, config)
        return "P5"

    def _classify_relation(
        self, rel_path: str, target_files: list[str], target_dirs: set[str]
    ) -> int:
        if rel_path in target_files:
            return 1  # exact match
        parent = str(Path(rel_path).parent)
        if parent in target_dirs:
            return 2  # same directory sibling
        return 3  # shared/other

    def _total_lines(self, candidates: list[_Candidate]) -> int:
        return sum(c.line_count for c in candidates)

    def _compute_hash(
        self, node_name: str, target_files: list[str], chunks: list[ContextChunk]
    ) -> str:
        manifest = {
            "node_name": node_name,
            "target_files": sorted(target_files),
            "chunks": [
                {
                    "path": c.path,
                    "priority": c.priority,
                    "start_line": c.start_line,
                    "end_line": c.end_line,
                    "content_sha256": c.content_sha256,
                }
                for c in chunks
            ],
        }
        canonical = json.dumps(manifest, sort_keys=True, ensure_ascii=False)
        return hashlib.sha256(canonical.encode()).hexdigest()[:16]
```
