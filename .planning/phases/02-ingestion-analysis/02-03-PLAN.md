```python
"""SemanticIntentExtractor — LLM-powered UI intent extraction (INGEST-04)."""

from __future__ import annotations

import base64
import hashlib
import json
from typing import Any

from ddk.adapters.llm.base import LLMAdapter, LLMCallConfig, LLMResponse, Message
from ddk.models.input import CaseInput
from ddk.models.intent import IntentGraph

# System prompt from docs/design/16-prompt-design.md §1
_SYSTEM_PROMPT = """You are an expert UI analyst. Your job is to analyze UI screenshots or mockups and extract a precise structural representation of the interface.

You must output a valid JSON object conforming to the IntentGraph schema.
Do not include any explanation outside the JSON.

## IntentGraph Rules

### LayoutRegion extraction
- Identify every distinct visual region: header, nav, sidebar, main content, footer, cards, modals, forms, tables, hero sections, CTAs, list views.
- Assign bounding boxes as relative coordinates (0.0 to 1.0).
- A region that contains others must list their IDs in `children`.
- `semantic_label` is a 1~5 word Korean or English label describing the region's purpose.

### ComponentSpec extraction
- Extract every interactive or semantic component: Button, Input, Card, Table, Badge, Icon, etc.
- `component_type` must be a PascalCase React-compatible component name.
- `props` should capture visible text content, variant hints, and disabled states.
- Assign `confidence` based on how clearly identifiable the component is (1.0 = clearly a known pattern).

### InteractionSpec extraction
- Only include interactions that are visually implied (buttons, links, form submits).
- `trigger` must be one of: click, hover, focus, submit, change, keydown, scroll.
- `action` must be one of: navigate, open-modal, close-modal, submit-form, toggle, filter, sort, load-more, other.

### Confidence scoring
- 1.0 = unambiguous standard component
- 0.7~0.9 = likely but uncertain
- <0.7 = low confidence, list in `unresolved_items`

## Critical constraints
- Do NOT hallucinate components that are not visible in the image.
- If an element is unclear, lower its confidence and add a description to `unresolved_items`.
- Maximum 50 components per image. If more are present, group minor ones."""


class SemanticIntentExtractor:
    """Extracts IntentGraph from UI screenshots via Claude vision (sonnet)."""

    def __init__(self, llm: LLMAdapter) -> None:
        self._llm = llm
        self._config = LLMCallConfig(
            model="claude-3-5-sonnet-20241022",
            temperature=0.0,
            max_tokens=4096,
        )

    async def extract(self, case_input: CaseInput) -> IntentGraph:
        """Extract UI intent from the first image in case_input.

        Args:
            case_input: Contains image_paths, prompt, case_id.

        Returns:
            Validated IntentGraph with layout, components, interactions.
        """
        image_path = case_input.image_paths[0]
        image_bytes = image_path.read_bytes()
        image_b64 = base64.standard_b64encode(image_bytes).decode()
        image_hash = hashlib.sha256(image_bytes).hexdigest()[:16]

        # Determine media type from extension
        suffix = image_path.suffix.lower()
        media_type_map = {".png": "image/png", ".jpg": "image/jpeg", ".jpeg": "image/jpeg", ".webp": "image/webp"}
        media_type = media_type_map.get(suffix, "image/png")

        user_content = (
            f"Analyze this UI image and extract the IntentGraph.\n\n"
            f"User's intent description:\n{case_input.prompt}"
        )
        if case_input.reference_notes:
            user_content += f"\n\nAdditional context: {case_input.reference_notes}"
        user_content += "\n\nOutput the complete IntentGraph JSON object."

        messages = [
            Message(role="system", content=_SYSTEM_PROMPT),
            Message(
                role="user",
                content=user_content,
                images=[(image_b64, media_type)],
            ),
        ]

        response: LLMResponse = await self._llm.complete_structured(
            messages, self._config, output_schema=IntentGraph
        )

        data: dict[str, Any] = json.loads(response.content)
        # Inject required fields that come from our context, not LLM
        data["id"] = case_input.case_id
        data["source_image_hash"] = image_hash
        # Ensure defaults for optional lists
        data.setdefault("layout_regions", [])
        data.setdefault("components", [])
        data.setdefault("interactions", [])
        data.setdefault("unresolved_items", [])
        data.setdefault("confidence", 0.5)

        return IntentGraph.model_validate(data)
```

```python
"""DesignSystemInferencer — LLM-powered design token inference (INGEST-05)."""

from __future__ import annotations

import base64
import hashlib
import json
from typing import Any

from ddk.adapters.llm.base import LLMAdapter, LLMCallConfig, LLMResponse, Message
from ddk.models.design_system import DesignSystem, InferredDesignToken
from ddk.models.input import CaseInput
from ddk.models.repo import DesignTokens, RepoProfile

_SYSTEM_PROMPT = """Analyze the provided UI screenshot and repository profile to infer the design system tokens.

You must output a valid JSON object conforming to the DesignSystem schema.

## Token Categories
- color: CSS color values (hex, rgb, hsl)
- spacing: rem/px values for margins and paddings
- typography: font sizes, weights, line heights
- border_radius: border radius values
- breakpoint: responsive breakpoint values

## Rules
1. Only infer tokens that are clearly visible in the screenshot.
2. Assign confidence based on how certain you are (1.0 = definite, 0.5 = uncertain).
3. primitive_components: List reusable UI primitive names visible (Button, Card, Badge, etc).
4. If the repo already has design tokens (from RepoProfile), extend rather than replace.
5. Output JSON only, no explanations."""


class DesignSystemInferencer:
    """Infers DesignSystem tokens from screenshots and repo analysis."""

    def __init__(self, llm: LLMAdapter) -> None:
        self._llm = llm
        self._config = LLMCallConfig(
            model="claude-3-5-sonnet-20241022",
            temperature=0.0,
            max_tokens=2048,
        )

    async def infer(
        self, case_input: CaseInput, repo_profile: RepoProfile
    ) -> DesignSystem:
        """Infer design system tokens from UI screenshot + repo profile.

        Args:
            case_input: Contains image_paths for visual analysis.
            repo_profile: Existing repo conventions for context.

        Returns:
            DesignSystem with inferred tokens.
        """
        image_path = case_input.image_paths[0]
        image_bytes = image_path.read_bytes()
        image_b64 = base64.standard_b64encode(image_bytes).decode()

        suffix = image_path.suffix.lower()
        media_type_map = {".png": "image/png", ".jpg": "image/jpeg", ".jpeg": "image/jpeg", ".webp": "image/webp"}
        media_type = media_type_map.get(suffix, "image/png")

        user_content = (
            f"Infer the design system from this UI screenshot.\n\n"
            f"Repository profile:\n"
            f"- Framework: {repo_profile.framework}\n"
            f"- CSS approach: {repo_profile.css_approach}\n"
            f"- Existing design tokens: {repo_profile.design_tokens.model_dump_json()}\n\n"
            f"Output the DesignSystem JSON."
        )

        messages = [
            Message(role="system", content=_SYSTEM_PROMPT),
            Message(role="user", content=user_content, images=[(image_b64, media_type)]),
        ]

        response: LLMResponse = await self._llm.complete_structured(
            messages, self._config, output_schema=DesignSystem
        )

        data: dict[str, Any] = json.loads(response.content)
        data["case_id"] = case_input.case_id
        data.setdefault("inferred_tokens", [])
        data.setdefault("primitive_components", [])

        return DesignSystem.model_validate(data)

    def enrich_profile(
        self, design_system: DesignSystem, repo_profile: RepoProfile
    ) -> RepoProfile:
        """Enrich repo_profile.design_tokens with inferred tokens.

        Does not modify GraphState — returns updated RepoProfile.
        """
        tokens = repo_profile.design_tokens
        for token in design_system.inferred_tokens:
            if token.category == "color":
                tokens.colors[token.name] = token.value
            elif token.category == "spacing":
                tokens.spacing[token.name] = token.value
            elif token.category == "typography":
                tokens.typography[token.name] = {"value": token.value}
            elif token.category == "border_radius":
                tokens.border_radius[token.name] = token.value
            elif token.category == "breakpoint":
                tokens.breakpoints[token.name] = token.value

        # Return new RepoProfile with enriched tokens
        return repo_profile.model_copy(update={"design_tokens": tokens})
```

```python
from tests.fakes.fake_llm_adapter import FakeLLMAdapter


@pytest.fixture
def mock_design_system_json() -> str:
    return json.dumps({
        "case_id": "will-be-overwritten",
        "inferred_tokens": [
            {"name": "primary", "value": "#3B82F6", "category": "color", "confidence": 0.95},

        ],
        "primitive_components": ["Button", "Card", "Badge"],
        "token_source": "llm",
    })


@pytest.fixture
def case_input(tmp_path: Path) -> CaseInput:
    img = tmp_path / "design.png"
    img.write_bytes(b"\x89PNG" + b"\x00" * 50)
    return CaseInput(
        case_id="C-001",
        image_paths=[img],
        prompt="쇼핑몰 페이지",
        repo_path=tmp_path,
    )


@pytest.fixture
def repo_profile() -> RepoProfile:
    return RepoProfile(framework="react")


class TestDesignSystemInferencer:
    async def test_infer_returns_design_system(
        self, case_input: CaseInput, repo_profile: RepoProfile, mock_design_system_json: str
    ) -> None:
        fake_llm = FakeLLMAdapter(responses={"analyze": mock_design_system_json})
        inferencer = DesignSystemInferencer(llm=fake_llm)
        result = await inferencer.infer(case_input, repo_profile)
        assert isinstance(result, DesignSystem)
        assert result.case_id == "C-001"
        assert len(result.inferred_tokens) == 3

    async def test_enrich_profile_colors(
        self, repo_profile: RepoProfile, mock_design_system_json: str
    ) -> None:
        ds = DesignSystem.model_validate(json.loads(mock_design_system_json) | {"case_id": "C-001"})
        inferencer = DesignSystemInferencer(llm=FakeLLMAdapter())
        enriched = inferencer.enrich_profile(ds, repo_profile)
        assert enriched.design_tokens.colors["primary"] == "#3B82F6"

    async def test_enrich_profile_spacing(
        self, repo_profile: RepoProfile, mock_design_system_json: str
    ) -> None:
        ds = DesignSystem.model_validate(json.loads(mock_design_system_json) | {"case_id": "C-001"})
        inferencer = DesignSystemInferencer(llm=FakeLLMAdapter())
        enriched = inferencer.enrich_profile(ds, repo_profile)
        assert enriched.design_tokens.spacing["sm"] == "0.5rem"

    async def test_enrich_profile_typography(
        self, repo_profile: RepoProfile, mock_design_system_json: str
    ) -> None:
        ds = DesignSystem.model_validate(json.loads(mock_design_system_json) | {"case_id": "C-001"})
        inferencer = DesignSystemInferencer(llm=FakeLLMAdapter())
        enriched = inferencer.enrich_profile(ds, repo_profile)
        assert "heading1" in enriched.design_tokens.typography

    async def test_model_is_sonnet(
        self, case_input: CaseInput, repo_profile: RepoProfile, mock_design_system_json: str
    ) -> None:
        fake_llm = FakeLLMAdapter(responses={"analyze": mock_design_system_json})
        inferencer = DesignSystemInferencer(llm=fake_llm)
        await inferencer.infer(case_input, repo_profile)
        assert fake_llm.call_log[0]["config"].model == "claude-3-5-sonnet-20241022"

    async def test_system_prompt_key_not_you(self) -> None:
        # Must start with different word than SemanticIntentExtractor to avoid FakeLLM collision
        first_word = _SYSTEM_PROMPT.split()[0].lower()
        assert first_word != "you"
        assert first_word == "analyze"

        result = await design_system_node(state, context=container)

        assert result["repo_profile"].design_tokens.colors.get("primary") == "#3B82F6"
```
